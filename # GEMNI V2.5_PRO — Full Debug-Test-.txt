class CoordinateTransform:
    @staticmethod
    def equatorial_to_cartesian(ra_deg: float, dec_deg: float,
                                dist_ly: float) -> Position:
        ra = math.radians(ra_deg)
        dec = math.radians(dec_deg)
        x = dist_ly * math.cos(dec) * math.cos(ra)
        y = dist_ly * math.cos(dec) * math.sin(ra)
        z = dist_ly * math.sin(dec)
        return Position(x_ly=x, y_ly=y, z_ly=z)

    @staticmethod
    def cartesian_to_equatorial(pos: Position) -> Tuple[float, float, float]:
        r = math.sqrt(pos.x_ly**2 + pos.y_ly**2 + pos.z_ly**2)
        if r == 0.0:
            return 0.0, 0.0, 0.0
        dec = math.degrees(math.asin(max(-1.0, min(1.0, pos.z_ly / r))))
        ra = math.degrees(math.atan2(pos.y_ly, pos.x_ly))
        if ra < 0.0:
            ra += 360.0
        return ra, dec, r

    @staticmethod
    def galactic_to_icrs(l_deg: float, b_deg: float,
                         dist_ly: float) -> Tuple[float, float, float]:
        l_r = math.radians(l_deg)
        b_r = math.radians(b_deg)
        cos_b = math.cos(b_r)
        sin_b = math.sin(b_r)
        ngp_ra = math.radians(192.85948)
        ngp_dec = math.radians(27.12825)
        l_ncp = math.radians(122.93192)
        sin_ngp = math.sin(ngp_dec)
        cos_ngp = math.cos(ngp_dec)
        sin_dec = sin_ngp * sin_b + cos_ngp * cos_b * math.cos(l_ncp - l_r)
        dec = math.degrees(math.asin(max(-1.0, min(1.0, sin_dec))))
        y_num = cos_b * math.sin(l_ncp - l_r)
        x_num = cos_ngp * sin_b - sin_ngp * cos_b * math.cos(l_ncp - l_r)
        ra = math.degrees(math.atan2(y_num, x_num)) + 192.85948
        ra = ra % 360.0
        return ra, dec, dist_ly

    @staticmethod
    def propagate_position(pos: Position, pm: ProperMotion,
                           delta_years: float) -> Position:
        if (pm.radial_velocity_km_s == 0.0
                and pm.pm_ra_mas_yr == 0.0
                and pm.pm_dec_mas_yr == 0.0):
            return Position(pos.x_ly, pos.y_ly, pos.z_ly)
        ra, dec, dist = CoordinateTransform.cartesian_to_equatorial(pos)
        if dist == 0.0:
            return Position(0.0, 0.0, 0.0)
        d_rad = pm.radial_velocity_km_s * delta_years / SPEED_OF_LIGHT_KM_S
        pm_ra_rad = math.radians(pm.pm_ra_mas_yr / 3_600_000.0)
        pm_dec_rad = math.radians(pm.pm_dec_mas_yr / 3_600_000.0)
        d_tan_ra = dist * pm_ra_rad * delta_years
        d_tan_dec = dist * pm_dec_rad * delta_years
        new_dist = dist + d_rad
        new_ra = ra + math.degrees(math.atan2(d_tan_ra, dist))
        new_dec = dec + math.degrees(math.atan2(d_tan_dec, dist))
        return CoordinateTransform.equatorial_to_cartesian(
            new_ra, new_dec, max(0.0, new_dist))

    @staticmethod
    def test_roundtrip() -> bool:
        test_cases = [
            (14.49, -60.83, 4.37),
            (266.417, -29.0, 26000.0),
            (0.0, 0.0, 100.0),
            (180.0, 45.0, 500.0),
            (359.9, -89.9, 1.0),
        ]
        for ra_in, dec_in, dist_in in test_cases:
            pos = CoordinateTransform.equatorial_to_cartesian(
                ra_in, dec_in, dist_in)
            ra_out, dec_out, dist_out = CoordinateTransform.cartesian_to_equatorial(pos)
            if abs(dist_in - dist_out) > 0.01:
                return False
            if abs(ra_in - ra_out) > 0.1 and abs(ra_in - ra_out) < 359.9:
                return False
            if abs(dec_in - dec_out) > 0.1:
                return False
        return True

def is_within_boundary(distance_ly: float,
                       max_boundary: float = MAX_BOUNDARY_LY) -> bool:
    if not math.isfinite(distance_ly):
        return False
    return 0.0 <= distance_ly <= max_boundary


def boundary_check(obj: AstronomicalObject,
                   max_boundary: float = MAX_BOUNDARY_LY) -> bool:
    return is_within_boundary(obj.distance_ly, max_boundary)

class SpatialCell:
    __slots__ = ("key", "objects")

    def __init__(self, key: Tuple[int, int, int]):
        self.key = key
        self.objects: List[AstronomicalObject] = []

    def add(self, obj: AstronomicalObject) -> None:
        self.objects.append(obj)

    def __len__(self) -> int:
        return len(self.objects)


class SpatialIndex:
    def __init__(self, cell_size_ly: float = 100.0):
        self.cell_size_ly = cell_size_ly
        self._cells: Dict[Tuple[int, int, int], SpatialCell] = {}
        self._obj_index: Dict[str, Tuple[int, int, int]] = {}
        self._total_objects: int = 0

    def _cell_key(self, x: float, y: float, z: float) -> Tuple[int, int, int]:
        return (
            math.floor(x / self.cell_size_ly),
            math.floor(y / self.cell_size_ly),
            math.floor(z / self.cell_size_ly),
        )

    def insert(self, obj: AstronomicalObject) -> None:
        key = self._cell_key(obj.position.x_ly, obj.position.y_ly,
                             obj.position.z_ly)
        if key not in self._cells:
            self._cells[key] = SpatialCell(key)
        self._cells[key].add(obj)
        self._obj_index[obj.obj_id] = key
        self._total_objects += 1

    def remove(self, obj_id: str) -> bool:
        key = self._obj_index.get(obj_id)
        if key is None:
            return False
        cell = self._cells.get(key)
        if cell is None:
            # Inconsistent state: obj_index says key exists but cell missing
            del self._obj_index[obj_id]
            return False
        before = len(cell.objects)
        cell.objects = [o for o in cell.objects if o.obj_id != obj_id]
        removed = before - len(cell.objects)
        if removed == 0:
            # Object not actually in cell — clean up stale index
            del self._obj_index[obj_id]
            return False
        self._total_objects -= removed
        del self._obj_index[obj_id]
        if len(cell.objects) == 0:
            del self._cells[key]
        return True

    def query_radius(self, cx: float, cy: float, cz: float,
                     radius_ly: float) -> List[AstronomicalObject]:
        results: List[AstronomicalObject] = []
        r2 = radius_ly * radius_ly
        for cell in self._cells.values():
            ck = cell.key
            cell_cx = (ck[0] + 0.5) * self.cell_size_ly
            cell_cy = (ck[1] + 0.5) * self.cell_size_ly
            cell_cz = (ck[2] + 0.5) * self.cell_size_ly
            dmax = self.cell_size_ly * 0.866025
            cell_dist = math.sqrt(
                (cell_cx - cx)**2 + (cell_cy - cy)**2 + (cell_cz - cz)**2
            )
            if cell_dist > radius_ly + dmax:
                continue
            for obj in cell.objects:
                dx = obj.position.x_ly - cx
                dy = obj.position.y_ly - cy
                dz = obj.position.z_ly - cz
                if dx*dx + dy*dy + dz*dz <= r2:
                    results.append(obj)
        return results

    def query_nearest(self, cx: float, cy: float, cz: float,
                      k: int = 1) -> List[AstronomicalObject]:
        scored: List[Tuple[float, AstronomicalObject]] = []
        for cell in self._cells.values():
            for obj in cell.objects:
                dx = obj.position.x_ly - cx
                dy = obj.position.y_ly - cy
                dz = obj.position.z_ly - cz
                scored.append((dx*dx + dy*dy + dz*dz, obj))
        scored.sort(key=lambda t: t[0])
        return [obj for _, obj in scored[:k]]

    def query_cell(self, key: Tuple[int, int, int]) -> List[AstronomicalObject]:
        cell = self._cells.get(key)
        return list(cell.objects) if cell else []

    def query_visible(self, cam_pos: Tuple[float, float, float],
                      view_radius_ly: float) -> List[AstronomicalObject]:
        return self.query_radius(cam_pos[0], cam_pos[1], cam_pos[2],
                                 view_radius_ly)

    def get_object_by_id(self, obj_id: str) -> Optional[AstronomicalObject]:
        key = self._obj_index.get(obj_id)
        if key is None:
            return None
        cell = self._cells.get(key)
        if cell is None:
            return None
        for obj in cell.objects:
            if obj.obj_id == obj_id:
                return obj
        return None

    def get_object_by_name(self, name: str) -> Optional[AstronomicalObject]:
        for cell in self._cells.values():
            for obj in cell.objects:
                if obj.name == name:
                    return obj
        return None

    @property
    def total_objects(self) -> int:
        return self._total_objects

    @property
    def total_cells(self) -> int:
        return len(self._cells)

    def __len__(self) -> int:
        return self._total_objects

class FloatingOrigin:
    def __init__(self, stability_threshold_ly: float = 50.0):
        self.origin_offset = Position(0.0, 0.0, 0.0)
        self._threshold = stability_threshold_ly

    def update(self, cam_world: Position) -> bool:
        dx = cam_world.x_ly - self.origin_offset.x_ly
        dy = cam_world.y_ly - self.origin_offset.y_ly
        dz = cam_world.z_ly - self.origin_offset.z_ly
        dist = math.sqrt(dx*dx + dy*dy + dz*dz)
        if dist > self._threshold:
            self.origin_offset = Position(cam_world.x_ly, cam_world.y_ly,
                                          cam_world.z_ly)
            return True
        return False

    def world_to_render(self, world_pos: Position) -> Position:
        return Position(
            world_pos.x_ly - self.origin_offset.x_ly,
            world_pos.y_ly - self.origin_offset.y_ly,
            world_pos.z_ly - self.origin_offset.z_ly,
        )

class GPUBackendType(Enum):
    NONE = auto()
    MODERNGL = auto()
    GODOT4 = auto()


class GPUBackend:
    def __init__(self):
        self.backend_type: GPUBackendType = GPUBackendType.NONE
        self.initialized: bool = False
        self._buffers: Dict[str, Any] = {}

    def init(self) -> bool:
        # Deliberately abstract — real allocation in ModernGL/Godot4 extension
        self.backend_type = GPUBackendType.NONE
        self.initialized = False
        return False

    def allocate_buffer(self, buffer_id: str, data: Any) -> bool:
        if not self.initialized:
            return False
        self._buffers[buffer_id] = data
        return True

    def release_buffer(self, buffer_id: str) -> None:
        self._buffers.pop(buffer_id, None)

    def status_report(self) -> Dict[str, Any]:
        return {
            "backend": self.backend_type.name,
            "initialized": self.initialized,
            "active_buffers": len(self._buffers),
        }

        return (
            math.floor(x / self.cell_size_ly),
            math.floor(y / self.cell_size_ly),
            math.floor(z / self.cell_size_ly),
        )

    def stage_cell_upload(self, cell_key: Tuple[int, int, int],
                          vertex_data: Any) -> bool:
        if cell_key in self.resident_cells:
            return False
        for s in self.staging_queue:
            if s.key == cell_key:
                return False
        sc = StreamingCell(cell_key, vertex_data)
        sc.obj_count = len(vertex_data) if hasattr(vertex_data, '__len__') else 0
        self.staging_queue.append(sc)
        return True

    def flush_staging(self, max_uploads_per_frame: int = 4) -> int:
        uploaded = 0
        while self.staging_queue and uploaded < max_uploads_per_frame:
            sc = self.staging_queue.pop(0)
            sc.resident = True
            sc.upload_ts = time.time()
            if self.gpu_backend.initialized:
                cell_id = f"cell_{sc.key[0]}_{sc.key[1]}_{sc.key[2]}"
                self.gpu_backend.allocate_buffer(cell_id, sc.vertex_data)
            self.resident_cells[sc.key] = sc
            uploaded += 1
        return uploaded

    def evict_distant_cells(self,
                            current_pos: Tuple[float, float, float]
                            ) -> List[Tuple[int, int, int]]:
        evicted: List[Tuple[int, int, int]] = []
        for key in list(self.resident_cells.keys()):
            cx = (key[0] + 0.5) * self.cell_size_ly
            cy = (key[1] + 0.5) * self.cell_size_ly
            cz = (key[2] + 0.5) * self.cell_size_ly
            d = math.sqrt(
                (cx - current_pos[0])**2 +
                (cy - current_pos[1])**2 +
                (cz - current_pos[2])**2
            )
            if d > self.max_resident_radius:
                self.resident_cells.pop(key)
                if self.gpu_backend.initialized:
                    cell_id = f"cell_{key[0]}_{key[1]}_{key[2]}"
                    self.gpu_backend.release_buffer(cell_id)
                evicted.append(key)
        return evicted

    @property
    def resident_count(self) -> int:
        return len(self.resident_cells)

    @property
    def staging_count(self) -> int:
        return len(self.staging_queue)

class CacheManager:
    def __init__(self, cache_dir: str = CACHE_DIR):
        self.cache_dir = cache_dir
        os.makedirs(cache_dir, exist_ok=True)

    def _cache_path(self, name: str, fmt: str = "json") -> str:
        return os.path.join(self.cache_dir, f"{name}.{fmt}")

    def save_json(self, name: str, data: Any,
                  metadata: Optional[Dict] = None) -> str:
        payload = {
            "cache_version": CACHE_VERSION,
            "timestamp": time.time(),
            "metadata": metadata or {},
            "data": data,
        }
        payload_json = json.dumps(payload, indent=2, default=str)
        checksum = hashlib.sha256(payload_json.encode()).hexdigest()
        payload["checksum"] = checksum
        path = self._cache_path(name, "json")
        try:
            with open(path, "w") as f:
                json.dump(payload, f, indent=2, default=str)
        except IOError as e:
            raise IOError(f"Cache write failed for {path}: {e}")
        return path

    def load_json(self, name: str,
                  max_age_hours: float = 24.0) -> Optional[Dict]:
        path = self._cache_path(name, "json")
        if not os.path.exists(path):
            return None
        try:
            with open(path, "r") as f:
                payload = json.load(f)
        except (json.JSONDecodeError, IOError) as e:
            print(f"[CACHE] Corrupt cache {path}: {e}")
            return None

        stored_checksum = payload.pop("checksum", None)
        if stored_checksum is not None:
            recomputed = hashlib.sha256(
                json.dumps(payload, indent=2, default=str).encode()
            ).hexdigest()
            if recomputed != stored_checksum:
                print(f"[CACHE] Checksum mismatch for {path} — rejecting.")
                return None

        if payload.get("cache_version") != CACHE_VERSION:
            print(f"[CACHE] Version mismatch for {path} — rejecting.")
            return None

        ts = payload.get("timestamp", 0)
        if (time.time() - ts) > max_age_hours * 3600:
            print(f"[CACHE] Stale cache {path} — older than {max_age_hours}h.")
            return None

        return payload

    def invalidate(self, name: str) -> bool:
        path = self._cache_path(name, "json")
        if os.path.exists(path):
            os.remove(path)
            return True
        return False

                luminosity_solar=s.get("luminosity_solar"),
                parent_id=None,
            )

            obj = SolarSystemBody(
                obj_id=obj_id, name=name, obj_type=obj_type,
                position=pos, spherical=sph,
                distance_ly=0.0, provenance=provenance,
                data_quality=dq, source=source,

                mass_kg=s.get("mass_kg"),
                radius_km=s.get("radius_km"),

                    orbital_elements.get("parent") if orbital_elements else None

    return objects

        self.mode = mode
        self.seed = seed
        self.spatial = SpatialIndex(cell_size_ly=100.0)
        self.streamer = StreamingManager(cell_size_ly=100.0)
        self.cache = CacheManager()
        self.gpu = GPUBackend()
        self.lod = LODSystem()
        self.origin = FloatingOrigin()
        self.sim_time: float = EPOCH_J2000
        self._objects: Dict[str, AstronomicalObject] = {}
        self._warnings: List[str] = []
        self._load_catalog()

    def _load_catalog(self) -> None:
        catalog_data: List[AstronomicalObject] = []

        # 1. Solar System
        solar_objs = build_solar_system_objects()
        catalog_data.extend(solar_objs)

        # 2. Curated extrasolar catalog
        extrasolar_objs = build_curated_extrasolar_catalog()
        catalog_data.extend(extrasolar_objs)

        # 3. Build lookup of host star IDs for exoplanet validation
        star_ids = {
            o.obj_id for o in catalog_data
            if o.obj_type == ObjectType.STAR
        }

        # 4. Insert objects with boundary enforcement
        boundary_rejected: List[str] = []
        for obj in catalog_data:
            # Solar System bodies (Sun, planets, moons, belts, ISM) are
            # at distance_ly=0 and live at the Solar System barycenter.
            # They are always in-bound by definition.
            is_solar = isinstance(obj, (SolarSystemBody, ProceduralPopulation,
                                        ISMRegion))
            # The Sun is a Star at distance 0 — treat it as solar system
            is_sun = (obj.obj_type == ObjectType.STAR
                      and obj.distance_ly == 0.0
                      and obj.name == "Sun")

            if is_solar or is_sun:
                self._insert_object(obj)
                continue

            # All other objects must pass the boundary check
            if not boundary_check(obj):
                boundary_rejected.append(
                    f"{obj.name} (dist={obj.distance_ly} ly)")
                continue

            self._insert_object(obj)

        if boundary_rejected:
            self._warnings.append(
                f"Boundary: {len(boundary_rejected)} objects rejected: "
                + "; ".join(boundary_rejected))

        # 5. Validate and link exoplanets to hosts
        self._link_exoplanets(star_ids)

        # 6. GPU init attempt
        self.gpu.init()

    def _insert_object(self, obj: AstronomicalObject) -> None:
        if obj.obj_id in self._objects:
            raise ValueError(f"Duplicate object ID: {obj.obj_id}")
        self._objects[obj.obj_id] = obj
        self.spatial.insert(obj)

    def _link_exoplanets(self, valid_host_ids: Set[str]) -> None:
        """Link exoplanets to their host stars.

        Reports warnings for missing or out-of-bound hosts.
        """
        for obj in list(self._objects.values()):
            if not isinstance(obj, Exoplanet):
                continue
            if not obj.host_id:
                self._warnings.append(
                    f"Exoplanet {obj.name} has no host_id")
                continue
            host = self._objects.get(obj.host_id)
            if host is None:
                self._warnings.append(
                    f"Exoplanet {obj.name} host '{obj.host_id}' "
                    f"not in active catalogue (may be out-of-bound)")
                continue
            # Copy host position/distance to planet
            obj.distance_ly = host.distance_ly
            obj.position = Position(
                host.position.x_ly, host.position.y_ly,
                host.position.z_ly
            )
            obj.spherical = SphericalPosition(
                host.spherical.ra_deg, host.spherical.dec_deg,
                host.spherical.distance_ly
            )
            # Link host -> child
            if obj.obj_id not in host.children_ids:
                host.children_ids.append(obj.obj_id)
            obj.parent_id = host.obj_id

    # ---- Godot Integration API ----

    def get_object(self, obj_id: str) -> Optional[Dict[str, Any]]:
        obj = self._objects.get(obj_id)
        return self._serialize_object(obj) if obj else None

    def get_object_by_name(self, name: str) -> Optional[Dict[str, Any]]:
        obj = self.spatial.get_object_by_name(name)
        return self._serialize_object(obj) if obj else None

    def query_radius(self, cx: float, cy: float, cz: float,
                     radius_ly: float) -> List[Dict[str, Any]]:
        return [self._serialize_object(o)
                for o in self.spatial.query_radius(cx, cy, cz, radius_ly)]

    def query_nearest(self, cx: float, cy: float, cz: float,
                      k: int = 1) -> List[Dict[str, Any]]:
        return [self._serialize_object(o)
                for o in self.spatial.query_nearest(cx, cy, cz, k)]

    def get_visible_objects(self, cam_world_pos: Tuple[float, float, float],
                            view_radius_ly: float) -> List[Dict[str, Any]]:
        self.origin.update(Position(*cam_world_pos))
        raw = self.spatial.query_visible(cam_world_pos, view_radius_ly)
        result = []
        for obj in raw:
            d = self._serialize_object(obj)
            rp = self.origin.world_to_render(obj.position)
            d["render_x"] = rp.x_ly
            d["render_y"] = rp.y_ly
            d["render_z"] = rp.z_ly
            lod = self.lod.compute_lod(obj.distance_ly)
            d["lod"] = lod.name
            d["point_size"] = self.lod.point_size_for_lod(lod)
            d["metadata_detail"] = self.lod.metadata_detail_for_lod(lod)
            result.append(d)
        return result

    def get_cell(self, key: Tuple[int, int, int]) -> List[Dict[str, Any]]:
        return [self._serialize_object(o)
                for o in self.spatial.query_cell(key)]

    def get_object_metadata(self, obj_id: str) -> Optional[Dict[str, Any]]:
        return self.get_object(obj_id)

    def get_simulation_time(self) -> float:
        return self.sim_time

    def set_simulation_time(self, t: float) -> None:
        self.sim_time = t

    def get_dataset_info(self) -> Dict[str, Any]:
        type_counts: Dict[str, int] = defaultdict(int)
        for obj in self._objects.values():
            type_counts[obj.obj_type.name] += 1
        return {
            "version": __version__,
            "mode": self.mode.name,
            "total_objects": len(self._objects),
            "spatial_cells": self.spatial.total_cells,
            "boundary_ly": MAX_BOUNDARY_LY,
            "type_counts": dict(type_counts),
            "gpu_status": self.gpu.status_report(),
            "provenance_note": (
                "Curated fallback data (magnitude-limited subset). "
                "Not a complete 2,000 ly census."
            ),
            "warnings": list(self._warnings),
        }

    def _serialize_object(self, obj: AstronomicalObject) -> Dict[str, Any]:
        d: Dict[str, Any] = {
            "id": obj.obj_id,
            "name": obj.name,
            "type": obj.obj_type.name,
            "x": round(obj.position.x_ly, 6),
            "y": round(obj.position.y_ly, 6),
            "z": round(obj.position.z_ly, 6),
            "ra": round(obj.spherical.ra_deg, 4),
            "dec": round(obj.spherical.dec_deg, 4),
            "dist": round(obj.distance_ly, 4),
            "mag": obj.apparent_mag,
            "provenance": obj.provenance.name,
            "quality": obj.data_quality.name,
            "source": obj.source,
        }
        if isinstance(obj, Star):
            d["spectral"] = obj.spectral_type
            d["temp_k"] = obj.temperature_k
            d["mass_solar"] = obj.mass_solar
        if isinstance(obj, WhiteDwarf):
            d["spectral"] = obj.spectral_type
            d["temp_k"] = obj.temperature_k
        if isinstance(obj, Exoplanet):
            d["host_id"] = obj.host_id
            d["host_name"] = obj.host_name
            d["period_days"] = obj.orbital_period_days
            d["mass_earth"] = obj.mass_earth
            d["method"] = obj.discovery_method
        if isinstance(obj, BlackHole):
            d["mass_solar"] = obj.mass_solar
            d["companion"] = obj.companion_name
        if isinstance(obj, SolarSystemBody):
            d["body_class"] = obj.body_class
            d["parent_body"] = obj.parent_body
        if isinstance(obj, ProceduralPopulation):
            d["population_type"] = obj.population_type
            d["seed"] = obj.seed
        if isinstance(obj, ISMRegion):
            d["region_type"] = obj.region_type
            d["extent_ly"] = obj.extent_ly
        return d

class ValidationResult:
    def __init__(self):
        self.total = 0
        self.passed = 0
        self.failed = 0
        self.errors: List[str] = []

    def ok(self, name: str) -> None:
        self.total += 1
        self.passed += 1
        print(f"  [PASS] {name}")

    def fail(self, name: str, reason: str) -> None:
        self.total += 1
        self.failed += 1
        msg = f"{name}: {reason}"
        self.errors.append(msg)
        print(f"  [FAIL] {msg}")

    def section(self, title: str) -> None:
        print(f"\n  -- {title} --")

    def summary(self) -> None:
        print(f"\n  Results: {self.passed}/{self.total} passed, "
              f"{self.failed} failed.")
        if self.errors:
            print("  Failures:")
            for e in self.errors:
                print(f"    - {e}")


def run_validation() -> bool:
    print(f"[VALIDATION] GEMNI V{__version__} Validation Suite")
    print("=" * 60)
    r = ValidationResult()

    engine = GEMNIEngine(mode=SimMode.OFFLINE)

    # -------------------------------------------------
    r.section("BOUNDARY — Hard 2,000 ly enforcement")
    # -------------------------------------------------
    def check_boundary(name: str, dist: float, expected: bool) -> None:
        actual = is_within_boundary(dist)
        if actual == expected:
            r.ok(f"{name} (dist={dist} ly, in_bound={expected})")
        else:
            r.fail(name, f"dist={dist}, expected in_bound={expected}, got {actual}")

    check_boundary("At boundary (2000.0)", 2000.0, True)
    check_boundary("Just inside (1999.99)", 1999.99, True)
    check_boundary("Just outside (2000.01)", 2000.01, False)
    check_boundary("At origin (0.0)", 0.0, True)
    check_boundary("Negative distance", -1.0, False)
    check_boundary("NaN distance", float('nan'), False)
    check_boundary("Positive infinity", float('inf'), False)
    check_boundary("Negative infinity", float('-inf'), False)

    # Verify every extrasolar object in the catalog obeys boundary
    boundary_ok = True
    boundary_violations = []
    for obj_id, obj in engine._objects.items():
        if obj.distance_ly == 0.0:
            continue  # Solar system / procedural — exempt
        if obj.obj_type == ObjectType.OTHER:
            continue  # ISM / procedural
        if not is_within_boundary(obj.distance_ly):
            boundary_violations.append(f"{obj.name} ({obj.distance_ly} ly)")
            boundary_ok = False
    if boundary_ok:
        r.ok("All active extrasolar objects satisfy distance <= 2000 ly")
    else:
        r.fail("Catalogue boundary", f"Violations: {boundary_violations}")

    # -------------------------------------------------
    r.section("EXCLUSIONS — Sgr A*, Carina, external objects")
    # -------------------------------------------------
    names = {o.name for o in engine._objects.values()}
    r.ok("Sgr A* excluded") if "Sagittarius A*" not in names else \
        r.fail("Sgr A*", "Found in active catalogue")
    r.ok("Carina Nebula excluded") if "Carina Nebula" not in names else \
        r.fail("Carina Nebula", "Found in active catalogue")

    # Test that boundary rejects a fake external star
    ext = Star(obj_id="EXT_TEST", name="External Star",
               obj_type=ObjectType.STAR, distance_ly=5000.0)
    r.ok("External 5000 ly object rejected") if not boundary_check(ext) else \
        r.fail("External object", "5000 ly star passed boundary")

    # -------------------------------------------------
    r.section("BLACK HOLES — Gaia BH1 / BH3")
    # -------------------------------------------------
    bh1 = engine._objects.get("BH_001")
    bh3 = engine._objects.get("BH_002")

    for bh, label, dist_expected in [
        (bh1, "Gaia BH1", 1560.0),
        (bh3, "Gaia BH3", 1926.0),
    ]:
        if bh is None:
            r.fail(f"{label} presence", "Not found in catalogue")
            continue
        r.ok(f"{label} present")
        if bh.obj_type != ObjectType.BLACK_HOLE:
            r.fail(f"{label} type", f"Expected BLACK_HOLE, got {bh.obj_type}")
        else:
            r.ok(f"{label} type = BLACK_HOLE")
        if not is_within_boundary(bh.distance_ly):
            r.fail(f"{label} boundary", f"dist={bh.distance_ly} > 2000")
        else:
            r.ok(f"{label} in boundary ({bh.distance_ly} ly)")
        if abs(bh.distance_ly - dist_expected) > 1.0:
            r.fail(f"{label} distance", f"Expected ~{dist_expected}, got {bh.distance_ly}")
        else:
            r.ok(f"{label} distance = {bh.distance_ly} ly")

    # -------------------------------------------------
    r.section("EXOPLANETS — Host correctness")
    # -------------------------------------------------
    prox_b = engine._objects.get("EXO_001")
    prox_d = engine._objects.get("EXO_002")

    if prox_b is None:
        r.fail("Proxima b presence", "EXO_001 not found")
    else:
        r.ok("Proxima Centauri b present")
        if prox_b.host_name == "Proxima Centauri":
            r.ok("Proxima b host_name = Proxima Centauri")
        else:
            r.fail("Proxima b host_name", f"Got '{prox_b.host_name}'")
        if prox_b.host_id == "STAR_003":
            r.ok("Proxima b host_id = STAR_003 (Proxima Centauri)")
        else:
            r.fail("Proxima b host_id", f"Got '{prox_b.host_id}'")
        if prox_b.host_id == "STAR_001":
            r.fail("Proxima b host",
                   "Still incorrectly linked to Alpha Centauri A!")
        else:
            r.ok("Proxima b NOT linked to Alpha Centauri A (correct)")
        # Check distance was propagated from host
        prox_star = engine._objects.get("STAR_003")
        if prox_star and abs(prox_b.distance_ly - prox_star.distance_ly) < 0.01:
            r.ok(f"Proxima b distance propagated from host "
                 f"({prox_b.distance_ly} ly)")
        else:
            r.fail("Proxima b distance", f"dist={prox_b.distance_ly}")

    if prox_d is None:
        r.fail("Proxima d presence", "EXO_002 not found")
    else:
        r.ok("Proxima Centauri d present")
        if prox_d.host_id == "STAR_003":
            r.ok("Proxima d host = Proxima Centauri")
        else:
            r.fail("Proxima d host", f"Got '{prox_d.host_id}'")

    # Invalid host test
    bad_exo = Exoplanet(obj_id="EXO_BAD", name="Bad Planet",
                        obj_type=ObjectType.PLANET,
                        host_id="NONEXISTENT")
    host = engine._objects.get(bad_exo.host_id)
    r.ok("Invalid host resolves to None") if host is None else \
        r.fail("Invalid host", "Unexpectedly resolved")

    # Duplicate planet deduplication
    exo_ids = [o.obj_id for o in engine._objects.values()
               if isinstance(o, Exoplanet)]
    if len(exo_ids) == len(set(exo_ids)):
        r.ok(f"No duplicate exoplanet IDs ({len(exo_ids)} planets)")
    else:
        r.fail("Duplicate exoplanets", "Duplicate IDs detected")

    # -------------------------------------------------
    r.section("PROVENANCE — All objects have explicit provenance")
    # -------------------------------------------------
    missing = [o.obj_id for o in engine._objects.values()
               if not isinstance(o.provenance, Provenance)]
    if not missing:
        r.ok(f"All {len(engine._objects)} objects have valid provenance")
    else:
        r.fail("Provenance coverage", f"{len(missing)} missing")

    masquerade = [o.obj_id for o in engine._objects.values()
                  if ("curated" in o.source.lower()
                      and o.provenance == Provenance.OBSERVED)]
    if not masquerade:
        r.ok("No curated data masquerading as OBSERVED")
    else:
        r.fail("Provenance honesty", f"{len(masquerade)} masquerading")

    # Check procedural objects are not labeled OBSERVED
    proc_as_obs = [o.obj_id for o in engine._objects.values()
                   if isinstance(o, (ProceduralPopulation, ISMRegion))
                   and o.provenance == Provenance.OBSERVED]
    if not proc_as_obs:
        r.ok("Procedural/ISM objects correctly not labeled OBSERVED")
    else:
        r.fail("Procedural provenance", f"{len(proc_as_obs)} mislabeled")

    # -------------------------------------------------
    r.section("COORDINATES — Transforms & round-trips")
    # -------------------------------------------------
    if CoordinateTransform.test_roundtrip():
        r.ok("Equatorial ↔ Cartesian round-trip (5 cases)")
    else:
        r.fail("Round-trip", "Error exceeded tolerance")

    # Galactic ↔ ICRS distance preservation
    _, _, d_out = CoordinateTransform.galactic_to_icrs(0.0, 0.0, 100.0)
    if abs(d_out - 100.0) < 0.01:
        r.ok("Galactic → ICRS distance preserved")
    else:
        r.fail("Galactic → ICRS", f"dist: 100 -> {d_out}")

    # Boundary round-trip
    pos_edge = CoordinateTransform.equatorial_to_cartesian(
        180.0, -45.0, MAX_BOUNDARY_LY)
    _, _, d_edge = CoordinateTransform.cartesian_to_equatorial(pos_edge)
    if abs(d_edge - MAX_BOUNDARY_LY) < 0.01:
        r.ok("Boundary distance preserved in coordinate transform")
    else:
        r.fail("Boundary coord", f"Expected {MAX_BOUNDARY_LY}, got {d_edge}")

    # Galactic pole test
    l, b = 0.0, 90.0
    ra_g, dec_g, d_g = CoordinateTransform.galactic_to_icrs(l, b, 50.0)
    if abs(d_g - 50.0) < 0.01:
        r.ok("Galactic pole distance preserved")
    else:
        r.fail("Galactic pole", f"dist: 50 -> {d_g}")

    # -------------------------------------------------
    r.section("SPATIAL INDEX — Insert, query, remove, negative coords")
    # -------------------------------------------------
    idx = SpatialIndex(cell_size_ly=50.0)
    t1 = Star(obj_id="T1", name="Test1", obj_type=ObjectType.STAR,
              position=Position(10.0, 20.0, 30.0), distance_ly=37.42)
    t2 = Star(obj_id="T2", name="Test2", obj_type=ObjectType.STAR,
              position=Position(-500.0, -300.0, -100.0), distance_ly=583.1)

    idx.insert(t1)
    idx.insert(t2)
    r.ok(f"Insert 2 objects (total={idx.total_objects})")

    found = idx.query_radius(10.0, 20.0, 30.0, 1.0)
    if any(o.obj_id == "T1" for o in found):
        r.ok("Radius query finds nearby object")
    else:
        r.fail("Radius query", "Nearby object not found")

    empty = idx.query_radius(10000.0, 10000.0, 10000.0, 1.0)
    if not any(o.obj_id == "T1" for o in empty):
        r.ok("Radius query returns empty for distant point")
    else:
        r.fail("Radius query (distant)", "False positive")

    nearest = idx.query_nearest(10.0, 20.0, 30.0, k=1)
    if nearest and nearest[0].obj_id == "T1":
        r.ok("Nearest query returns correct object")
    else:
        r.fail("Nearest query", "Wrong object")

    neg_found = idx.query_radius(-500.0, -300.0, -100.0, 5.0)
    if any(o.obj_id == "T2" for o in neg_found):
        r.ok("Negative-coordinate cell query works")
    else:
        r.fail("Negative coordinates", "Object not found")

    # Remove and verify
    idx.remove("T1")
    r.ok("Object removed") if idx.get_object_by_id("T1") is None else \
        r.fail("Remove", "Object still present")

    # Double-remove (should not crash or go negative)
    removed_again = idx.remove("T1")
    r.ok("Double-remove returns False safely") if not removed_again else \
        r.fail("Double-remove", "Returned True for already-removed object")
    if idx.total_objects >= 0:
        r.ok(f"Count non-negative after double-remove ({idx.total_objects})")
    else:
        r.fail("Count", f"Negative count: {idx.total_objects}")

    # -------------------------------------------------
    r.section("STREAMING — Stage, flush, eviction, duplicates, pressure")
    # -------------------------------------------------
    sm = StreamingManager(cell_size_ly=100.0)
    key = sm.compute_cell_key(10.0, 20.0, 30.0)

    r.ok("Cell staged") if sm.stage_cell_upload(key, [0.0]) else \
        r.fail("Stage", "Failed")
    r.ok("Duplicate rejected") if not sm.stage_cell_upload(key, [0.0]) else \
        r.fail("Duplicate", "Accepted")

    flushed = sm.flush_staging(max_uploads_per_frame=4)
    if key in sm.resident_cells and flushed == 1:
        r.ok(f"Flush: {flushed} uploaded, {sm.resident_count} resident")
    else:
        r.fail("Flush", f"flushed={flushed}, resident={sm.resident_count}")

    evicted = sm.evict_distant_cells((100000.0, 100000.0, 100000.0))
    if key not in sm.resident_cells and key in evicted:
        r.ok("Distant cell evicted")
    else:
        r.fail("Eviction", "Cell not evicted")

    # Queue pressure
    for i in range(10):
        sm.stage_cell_upload((i, 0, 0), [float(i)])
    r.ok(f"Queue pressure: {sm.staging_count} staged") \
        if sm.staging_count == 10 else \
        r.fail("Queue", f"Expected 10, got {sm.staging_count}")

    sm.flush_staging(max_uploads_per_frame=4)
    r.ok(f"Batched flush: 4 of 10, resident={sm.resident_count}") \
        if sm.resident_count == 4 else \
        r.fail("Batch", f"resident={sm.resident_count}")

    # -------------------------------------------------
    r.section("LOD — Deterministic 5-band assignment")
    # -------------------------------------------------
    lod = LODSystem()
    lod_tests = [
        (5.0, LODLevel.LOD_0), (10.0, LODLevel.LOD_0),
        (10.1, LODLevel.LOD_1), (100.0, LODLevel.LOD_1),
        (100.1, LODLevel.LOD_2), (500.0, LODLevel.LOD_2),
        (500.1, LODLevel.LOD_3), (1000.0, LODLevel.LOD_3),
        (1000.1, LODLevel.LOD_4), (2000.0, LODLevel.LOD_4),
    ]
    all_lod_ok = True
    for dist, expected in lod_tests:
        actual = lod.compute_lod(dist)
        if actual != expected:
            r.fail("LOD", f"dist={dist}: expected {expected}, got {actual}")
            all_lod_ok = False
    if all_lod_ok:
        r.ok("LOD deterministic for all 10 boundary test points")

    # Point sizes monotonically decrease
    sizes = [lod.point_size_for_lod(l) for l in LODLevel]
    if all(sizes[i] >= sizes[i+1] for i in range(len(sizes)-1)):
        r.ok("LOD point sizes monotonically decrease")
    else:
        r.fail("LOD sizes", f"Not monotonic: {sizes}")

    # -------------------------------------------------
    r.section("CACHE — Save, load, corruption, version mismatch")
    # -------------------------------------------------
    cm = CacheManager(cache_dir=".gemni_test_cache")
    test_data = {"objects": [{"id": "t1"}], "count": 1}
    meta = {"source": "unit_test"}

    path = cm.save_json("test", test_data, metadata=meta)
    r.ok("Cache saved")

    loaded = cm.load_json("test", max_age_hours=1.0)
    if loaded and loaded.get("data", {}).get("count") == 1:
        r.ok("Cache loaded, data round-trip correct")
    else:
        r.fail("Cache load", "Data mismatch")

    # Corrupt the file
    with open(path, "r") as f:
        content = f.read()
    with open(path, "w") as f:
        f.write(content[:-20] + "CORRUPTED_DATA!!!")

    corrupt_result = cm.load_json("test", max_age_hours=1.0)
    r.ok("Corrupted cache rejected") if corrupt_result is None else \
        r.fail("Corruption", "Corrupted cache accepted")

    # Cleanup
    cm.invalidate("test")
    r.ok("Cache invalidated") if not os.path.exists(path) else \
        r.fail("Invalidate", "File persists")
    try:
        os.rmdir(".gemni_test_cache")
    except OSError:
        pass

    # -------------------------------------------------
    r.section("FLOATING ORIGIN — Shift, render coordinates")
    # -------------------------------------------------
    fo = FloatingOrigin(stability_threshold_ly=50.0)
    shifted = fo.update(Position(100.0, 100.0, 100.0))
    r.ok("Origin shifts for large displacement") if shifted else \
        r.fail("Shift", "Did not shift")

    rp = fo.world_to_render(Position(100.0, 100.0, 100.0))
    if abs(rp.x_ly) < 0.01 and abs(rp.y_ly) < 0.01 and abs(rp.z_ly) < 0.01:
